12. Return & Rekursive Funksjoner

return sender en verdi ut av funksjonen slik at svaret kan lagres i en variabel, og avslutter funksjonen umiddelbart.

function calculator(number1, number2, operator) {
    if (operator === "add")      return number1 + number2;
    if (operator === "subtract") return number1 - number2;
    if (operator === "multiply") return number1 * number2;
    if (operator === "divide")   return number1 / number2;
    return; // Avslutter funksjonen hvis operator mangler/er feil
}

// Lagre resultatet i en variabel og skriv ut med createElement:
const result = calculator(2, 3, "add"); // 5
const p = document.createElement("p");
p.innerHTML = result;
document.body.appendChild(p);

Rekursiv funksjon (funksjon som kaller seg selv)

function countdown(from) {
    if (from > 0) {
        console.log(from); // 1. Gjør arbeidet
        from--;            // 2. Endre verdien mot stoppbetingelsen
        countdown(from);   // 3. Kall seg selv på nytt
    } else {
        return;            // 4. Stoppbetingelse (viktig for å unngå evig løkke!)
    }
}
countdown(10);

let direction = "";

// Lytt etter når en tast trykkes NED på hele dokumentet:
document.addEventListener("keydown", function(event) {
    // event.key = tegnet ("a", "w", "ArrowUp", "Enter")
    // event.code = fysisk tastekode ("KeyA", "KeyW", "Space")
    if (event.key === "a") direction = "left";

    if (event.key === "s") direction = "down";

});

// Lytt etter når en tast slippes OPP:
document.addEventListener("keyup", function(event) {
    if (event.key === "a" || event.key === "w" || event.key === "s" || event.key === "d") {
        direction = "";
        pd.innerHTML = direction;
    }
});

  • "2" + "3" = "23" (Tall fra input er alltid String!): Verdier hentet med .value er tekst. Skal du plusse, bruk Number(document.getElementById("id").value)!
  • Siden laster på nytt og tømmer alt ved skjemainnsending: Du mangler event.preventDefault() inne i addEventListener("submit", function(event) { ... }).
  • Feilmelding Cannot set properties of null (setting 'innerHTML'): ID-en i document.getElementById("...") matcher ikke id="..." i HTML (sjekk store/små bokstaver!), eller <script> ligger før HTML-elementet.
  • Kun siste element vises i listen: Du har skrevet innerHTML = ... i stedet for innerHTML += ... inne i .map() / løkka.
  • Listen dupliserer seg hver gang man legger til noe nytt: Tøm HTML-elementet eller bruk let html = "" i starten av utskriftsfunksjonen før .map().
  • = vs ===: Ett likhetstegn (=) setter en ny verdi. Tre likhetstegn (===) sammenligner to verdier i en if-test.
  • Tallsortering blir feil (100 kommer før 20): Husk sammenligningsfunksjon: tall.sort((a, b) => a - b).
